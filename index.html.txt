    <h1>商品価格・割引チェック（テスト版）</h1>
    
    <!-- カメラ画面 -->
    <div id="reader"></div>

    <!-- スキャン結果表示エリア -->
    <div id="result" class="result-box">
        <p><strong>商品名:</strong> <span id="p-name">-</span></p>
        <p><strong>定価:</strong> <span id="p-list">-</span> 円</p>
        <p><strong>今週の売価:</strong> <span id="p-sale" class="price">-</span> 円</p>
        <p><strong>割引率:</strong> <span id="p-disc" class="discount">-</span></p>
    </div>

    <script>
        // 先ほどコピーしたご自身のGAS（ウェブアプリ）URLに書き換えてください
        const GAS_URL = "ここにGASのウェブアプリURLを貼り付けてください";

        let productData = [];

        // スプレッドシートからのデータ取得
        async function loadData() {
            try {
                const response = await fetch(GAS_URL);
                productData = await response.json();
                console.log("データ読み込み成功:", productData.length + "件");
            } catch (error) {
                alert("商品データの読み込みに失敗しました。GASのURLを確認してください。");
            }
        }
        loadData();

        // バーコード検出時の処理
        function onScanSuccess(decodedText) {
            const item = productData.find(p => String(p.jan) === String(decodedText));

            const resultBox = document.getElementById("result");
            if (item) {
                document.getElementById("p-name").textContent = item.name;
                document.getElementById("p-list").textContent = item.list_price ? Number(item.list_price).toLocaleString() : "-";
                document.getElementById("p-sale").textContent = item.sale_price ? Number(item.sale_price).toLocaleString() : "-";
                document.getElementById("p-disc").textContent = item.discount_rate || "-";
                resultBox.style.display = "block";
            } else {
                alert("該当する商品が見つかりませんでした (JAN: " + decodedText + ")");
            }
        }

        // カメラ起動
        const html5QrCode = new Html5Qrcode("reader");
        html5QrCode.start(
            { facingMode: "environment" },
            { fps: 10, qrbox: { width: 250, height: 150 } },
            onScanSuccess
        ).catch(err => {
            console.error("カメラ起動エラー", err);
        });
    </script>
